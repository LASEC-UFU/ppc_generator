' =============================================================================
' FORMULÁRIO DE SELEÇÃO MÚLTIPLA DE COMPONENTES
'
' Instalação:
'   1. Importe este módulo no VBA.
'   2. Execute InstalarFormularioComponentes uma única vez.
'   3. Associe um botão à macro AbrirFormularioComponentes.
'
' Uso:
'   - Selecione uma célula abaixo de um cabeçalho chamado componentes.
'   - Abra o formulário.
'   - Pesquise e marque vários nomes.
'   - O campo recebe os códigos separados por |.
'   - Códigos que já estavam no campo aparecem previamente marcados.
' =============================================================================

Private Sub RemoverComponenteSeExistir(ByVal projeto As Object, ByVal nome As String)
    Dim componente As Object

    On Error Resume Next
    Set componente = projeto.VBComponents(nome)
    On Error GoTo 0

    If Not componente Is Nothing Then
        projeto.VBComponents.Remove componente
    End If
End Sub

' -----------------------------------------------------------------------------
' ABERTURA DO FORMULÁRIO
' -----------------------------------------------------------------------------
Public Sub AbrirFormularioComponentes()
    Dim alvo As Range
    Dim formulario As Object

    On Error GoTo TratarErro

    If TypeName(Selection) <> "Range" Then
        Err.Raise vbObjectError + 3100, "AbrirFormularioComponentes", _
                  "Selecione uma célula da coluna componentes."
    End If

    If Selection.Cells.CountLarge <> 1 Then
        Err.Raise vbObjectError + 3101, "AbrirFormularioComponentes", _
                  "Selecione somente uma célula."
    End If

    Set alvo = Selection.Cells(1, 1)

    If alvo.MergeCells Then
        Set alvo = alvo.MergeArea.Cells(1, 1)
    End If

    If Not EhCelulaDaColunaComponentesFormulario(alvo) Then
        Err.Raise vbObjectError + 3102, "AbrirFormularioComponentes", _
                  "A célula " & alvo.Worksheet.Name & "!" & alvo.Address(False, False) & _
                  " não está abaixo de um cabeçalho chamado componentes."
    End If

    Set gCelulaDestinoFormulario = alvo
    Set formulario = VBA.UserForms.Add(NOME_FORMULARIO)
    formulario.Show
    Set formulario = Nothing
    Exit Sub

TratarErro:
    MsgBox "Não foi possível abrir o formulário." & vbNewLine & vbNewLine & _
           Err.Number & " - " & Err.Description, _
           vbExclamation, "Selecionar componentes"
End Sub

Public Function ObterCelulaDestinoFormulario() As Range
    Set ObterCelulaDestinoFormulario = gCelulaDestinoFormulario
End Function

Public Sub LimparDestinoFormulario()
    Set gCelulaDestinoFormulario = Nothing
End Sub

Public Function EhCelulaDaColunaComponentesFormulario(ByVal alvo As Range) As Boolean
    EhCelulaDaColunaComponentesFormulario = _
        (LocalizarLinhaCabecalhoFormulario(alvo) > 0)
End Function

Private Function LocalizarLinhaCabecalhoFormulario(ByVal alvo As Range) As Long
    Dim linha As Long

    If alvo Is Nothing Then Exit Function
    If Not (alvo.Parent.Parent Is ThisWorkbook) Then Exit Function

    For linha = alvo.Row To 1 Step -1
        If NormalizarCabecalhoFormulario( _
            alvo.Worksheet.Cells(linha, alvo.Column).Value2) = CABECALHO_ALVO Then

            If linha < alvo.Row Then
                LocalizarLinhaCabecalhoFormulario = linha
                Exit Function
            End If
        End If
    Next linha
End Function

' -----------------------------------------------------------------------------
' FONTE DOS COMPONENTES
' -----------------------------------------------------------------------------
Public Function CarregarComponentesFormulario( _
    ByRef codigos() As String, _
    ByRef nomes() As String) As Long

    Dim ws As Worksheet
    Dim linhaCabecalho As Long
    Dim colunaCodigo As Long
    Dim colunaNome As Long
    Dim ultimaLinha As Long
    Dim linha As Long
    Dim total As Long
    Dim codigo As String
    Dim nome As String
    Dim usados As Object

    Set ws = ThisWorkbook.Worksheets(ABA_COMPONENTES)

    LocalizarCabecalhosFonteFormulario _
        ws, linhaCabecalho, colunaCodigo, colunaNome

    ultimaLinha = Application.Max( _
        ws.Cells(ws.Rows.Count, colunaCodigo).End(xlUp).Row, _
        ws.Cells(ws.Rows.Count, colunaNome).End(xlUp).Row)

    Set usados = CreateObject("Scripting.Dictionary")
    usados.CompareMode = vbTextCompare

    For linha = linhaCabecalho + 1 To ultimaLinha
        codigo = Trim$(CStr(ws.Cells(linha, colunaCodigo).Value2))
        nome = Trim$(CStr(ws.Cells(linha, colunaNome).Value2))

        If codigo <> vbNullString Or nome <> vbNullString Then
            If codigo = vbNullString Then
                Err.Raise vbObjectError + 3110, "CarregarComponentesFormulario", _
                          "Componente sem código em " & ws.Name & "!" & _
                          ws.Cells(linha, colunaCodigo).Address(False, False) & "."
            End If

            If nome = vbNullString Then
                Err.Raise vbObjectError + 3111, "CarregarComponentesFormulario", _
                          "Componente sem nome em " & ws.Name & "!" & _
                          ws.Cells(linha, colunaNome).Address(False, False) & "."
            End If

            If usados.Exists(codigo) Then
                Err.Raise vbObjectError + 3112, "CarregarComponentesFormulario", _
                          "Código duplicado na aba Componentes: " & codigo & "."
            End If

            usados.Add codigo, True
            total = total + 1

            If total = 1 Then
                ReDim codigos(1 To 1)
                ReDim nomes(1 To 1)
            Else
                ReDim Preserve codigos(1 To total)
                ReDim Preserve nomes(1 To total)
            End If

            codigos(total) = codigo
            nomes(total) = nome
        End If
    Next linha

    If total = 0 Then
        Err.Raise vbObjectError + 3113, "CarregarComponentesFormulario", _
                  "Nenhum componente foi encontrado na aba Componentes."
    End If

    CarregarComponentesFormulario = total
End Function

Private Sub LocalizarCabecalhosFonteFormulario( _
    ByVal ws As Worksheet, _
    ByRef linhaCabecalho As Long, _
    ByRef colunaCodigo As Long, _
    ByRef colunaNome As Long)

    Dim primeiraLinha As Long
    Dim ultimaLinhaBusca As Long
    Dim primeiraColuna As Long
    Dim ultimaColuna As Long
    Dim linha As Long
    Dim coluna As Long
    Dim cabecalho As String

    primeiraLinha = ws.UsedRange.Row
    ultimaLinhaBusca = Application.Min( _
        ws.UsedRange.Row + ws.UsedRange.Rows.Count - 1, primeiraLinha + 100)
    primeiraColuna = ws.UsedRange.Column
    ultimaColuna = ws.UsedRange.Column + ws.UsedRange.Columns.Count - 1

    For linha = primeiraLinha To ultimaLinhaBusca
        colunaCodigo = 0
        colunaNome = 0

        For coluna = primeiraColuna To ultimaColuna
            cabecalho = NormalizarCabecalhoFormulario( _
                ws.Cells(linha, coluna).Value2)

            Select Case cabecalho
                Case CABECALHO_CODIGO
                    colunaCodigo = coluna
                Case CABECALHO_NOME
                    colunaNome = coluna
            End Select
        Next coluna

        If colunaCodigo > 0 And colunaNome > 0 Then
            linhaCabecalho = linha
            Exit Sub
        End If
    Next linha

    Err.Raise vbObjectError + 3114, "LocalizarCabecalhosFonteFormulario", _
              "Não foram encontrados, na mesma linha da aba Componentes, " & _
              "os cabeçalhos codigo e nome."
End Sub

' -----------------------------------------------------------------------------
' NORMALIZAÇÃO
' -----------------------------------------------------------------------------
Private Function NormalizarCabecalhoFormulario(ByVal valor As Variant) As String
    Dim texto As String

    If IsError(valor) Or IsEmpty(valor) Or IsNull(valor) Then Exit Function

    texto = NormalizarPesquisaFormulario(CStr(valor))
    texto = Replace(texto, "-", "_")
    texto = Replace(texto, " ", "_")

    Do While InStr(texto, "__") > 0
        texto = Replace(texto, "__", "_")
    Loop

    NormalizarCabecalhoFormulario = texto
End Function

Public Function NormalizarPesquisaFormulario(ByVal valor As Variant) As String
    Dim texto As String

    If IsError(valor) Or IsEmpty(valor) Or IsNull(valor) Then Exit Function

    texto = LCase$(Trim$(CStr(valor)))
    texto = Replace(texto, ChrW(160), " ")
    texto = Replace(texto, "á", "a")
    texto = Replace(texto, "à", "a")
    texto = Replace(texto, "â", "a")
    texto = Replace(texto, "ã", "a")
    texto = Replace(texto, "ä", "a")
    texto = Replace(texto, "é", "e")
    texto = Replace(texto, "ê", "e")
    texto = Replace(texto, "ë", "e")
    texto = Replace(texto, "í", "i")
    texto = Replace(texto, "ï", "i")
    texto = Replace(texto, "ó", "o")
    texto = Replace(texto, "ô", "o")
    texto = Replace(texto, "õ", "o")
    texto = Replace(texto, "ö", "o")
    texto = Replace(texto, "ú", "u")
    texto = Replace(texto, "ü", "u")
    texto = Replace(texto, "ç", "c")

    Do While InStr(texto, "  ") > 0
        texto = Replace(texto, "  ", " ")
    Loop

    NormalizarPesquisaFormulario = texto
End Function
